92 size_t getColumn(double x) const
93 {
94 if (extent_tag::padding) {
95 if (x < m_extent.getMinX())
96 return 0;
97 if (x > m_extent.getMaxX())
98 return m_num_cols - 1;
99 if (x == m_extent.getMaxX()) // special case, returning the cell for which xmax is the right
100 return m_num_cols - 2;
101 } else {
102 if (x < m_extent.getMinX() || x > m_extent.getMaxX())
103 throw std::out_of_range("x");
104
105 if (x == m_extent.getMaxX())
106 return m_num_cols - 1;
107 }
108
109 // Since we've already range-checked our x value, make sure that
110 // the computed column index is no greater than the column index
111 // associated with xmax.
112 return std::min(
113 extent_tag::padding + static_cast<size_t>(std::floor((x - m_extent.getMinX()) / m_dx)),
114 getColumn(m_extent.getMaxX()));
115 }

137 // associated with ymin.
138 return std::min(

201 Grid<extent_tag> shrinkToFit(const geom::Envelope& b, bool calcExtentFromNewGrid=true) const
202 {
203 if (b.getArea() == 0) {
204 return makeEmpty();
205 }
206
207 if (b.getMinX() < m_extent.getMinX() || b.getMinY() < m_extent.getMinY() || b.getMaxX() > m_extent.getMaxX() || b.getMaxY() > m_extent.getMaxY()) {
208 throw std::range_error("Cannot shrink extent to bounds larger than original.");
209 }
210
211 size_t col0 = getColumn(b.getMinX());
212 size_t row1 = getRow(b.getMaxY());
213
214 // Shrink xmin and ymax to fit the upper-left corner of the supplied extent
215 double snapped_xmin = m_extent.getMinX() + static_cast<double>(col0 - extent_tag::padding) * m_dx;
216 double snapped_ymax = m_extent.getMaxY() - static_cast<double>(row1 - extent_tag::padding) * m_dy;
217
218 // Make sure snapped_xmin and snapped_ymax are within the reduced extent.
219 // Because of floating point round-off errors, this is not always the case.
220 if (b.getMinX() < snapped_xmin) {
221 snapped_xmin -= m_dx;
222 col0--;
223 }
224
225 if (b.getMaxY() > snapped_ymax) {
226 snapped_ymax += m_dy;
227 row1--;
228 }
229
230 size_t col1 = getColumn(b.getMaxX());
231 size_t row0 = getRow(b.getMinY());
232
233 size_t num_rows = 1 + (row0 - row1);
234 size_t num_cols = 1 + (col1 - col0);
235
236 // If xmax or ymin falls cleanly on a cell boundary, we don't
237 // need as many rows or columns as we otherwise would, because
238 // we assume that the rightmost cell of the grid is a closed
239 // interval in x, and the lowermost cell of the grid is a
240 // closed interval in y.
241 if (num_rows > 2 && (snapped_ymax - static_cast<double>(num_rows - 1) * m_dy <= b.getMinY())) {
242 num_rows--;
243 }
244 if (num_cols > 2 && (snapped_xmin + static_cast<double>(num_cols - 1) * m_dx >= b.getMaxX())) {
245 num_cols--;
246 }
247
248 const double snapped_xmax = m_extent.getMinX() + static_cast<double>(col0 + num_cols - extent_tag::padding) * m_dx;
249 const double snapped_ymin = m_extent.getMaxY() - static_cast<double>(row1 + num_rows - extent_tag::padding) * m_dy;
250
251 // Perform offsets relative to the new xmin, ymax origin
252 // points. If this is not done, then floating point roundoff
253 // error can cause progressive shrink() calls with the same
254 // inputs to produce different results.
255 geom::Envelope reduced_box = {
256 snapped_xmin,
257 calcExtentFromNewGrid ? std::max(snapped_xmin + static_cast<double>(num_cols) * m_dx, b.getMaxX()) : snapped_xmax,
258 calcExtentFromNewGrid ? std::min(snapped_ymax - static_cast<double>(num_rows) * m_dy, b.getMinY()) : snapped_ymin,
259 snapped_ymax
260 };
261
262 // Fudge computed xmax and ymin, if needed, to prevent extent
263 // from growing during a shrink operation.
264 if (reduced_box.getMaxX() > m_extent.getMaxX()) {
265 if (std::round(reduced_box.getWidth() / m_dx) ==
266 std::round((m_extent.getMaxX() - reduced_box.getMinX()) / m_dx)) {
267 reduced_box = geom::Envelope(reduced_box.getMinX(), m_extent.getMaxX(), reduced_box.getMinY(), reduced_box.getMaxY());
268 } else {
269 throw std::runtime_error("Shrink operation failed.");
270 }
271 }
272 if (reduced_box.getMinY() < m_extent.getMinY()) {
273 if (std::round(reduced_box.getHeight() / m_dy) ==
274 std::round((reduced_box.getMaxY() - m_extent.getMinY()) / m_dy)) {
275 reduced_box = geom::Envelope(reduced_box.getMinX(), reduced_box.getMaxX(), m_extent.getMinY(), reduced_box.getMaxY());
276 } else {
277 throw std::runtime_error("Shrink operation failed.");
278 }
279 }
280
281 if constexpr (std::is_same_v<extent_tag, bounded_extent>) {
282 Grid<extent_tag> reduced{reduced_box, m_dx, m_dy};
283
284 if (!calcExtentFromNewGrid) {
285 reduced.m_xOrig = m_xOrig;
286 reduced.m_yOrig = m_yOrig;
287 reduced.m_skipRows = reduced.getRowOffset(*this);
288 reduced.m_skipCols = reduced.getColOffset(*this);
289 } else if (!reduced.getExtent().contains(b)) {
290 throw std::runtime_error("Shrink operation failed.");
291 }
292 return reduced;
293 } else {
294 Grid<extent_tag> reduced{reduced_box, m_dx, m_dy, m_domain};
295
296 if (!calcExtentFromNewGrid) {
297 reduced.m_xOrig = m_xOrig;
298 reduced.m_yOrig = m_yOrig;
299 reduced.m_skipRows = reduced.getRowOffset(*this);
300 reduced.m_skipCols = reduced.getColOffset(*this);
301 }else if (!reduced.getExtent().contains(b)) {
302 throw std::runtime_error("Shrink operation failed.");
303 }
304 return reduced;
305 }
